# TempleOS Archive Transcriber - Python dependencies
# Install:  pip install -r requirements.txt
#
# Also required (not pip-installable):
#   ffmpeg          # brew install ffmpeg / sudo pacman -S ffmpeg / sudo apt install ffmpeg
#   ia configure    # optional: archive.org S3 keys for higher rate limits
#
# Optional recovery tooling (auto-detected; only needed to recover corrupt sources):
#   untrunc         # rebuilds truncated MP4s missing their moov index.  Easiest via
#                   #   docker:  git clone https://github.com/anthwlock/untrunc /tmp/untrunc
#                   #            docker build -t untrunc-local /tmp/untrunc
#                   #   (or install an `untrunc` binary on PATH)

# Archive.org item access
internetarchive[speedups]

# Re-fetch the original audio from YouTube for files that still carry a video id
# (recovery tier 3).  Optional - recovery degrades gracefully without it.
yt-dlp
